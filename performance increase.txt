const pLimit = require('p-limit');

const fetchInboxMailList = async (imap, results, folder) => {
  const limit = pLimit(10); // Limit concurrency to 10
  const fetchPromises = results.map((result) => 
    limit(() => fetchEmail(imap, result, folder))
  );

  let fetchedMails = await Promise.all(fetchPromises);
  return fetchedMails;
};

const fetchEmail = (imap, result, folder) => {
  return new Promise((resolve, reject) => {
    const f = imap.fetch(result, { bodies: "", struct: true });

    f.on("message", (msg) => {
      let rawEmail = "";
      let attributes = null;

      msg.on("body", (stream) => {
        stream.on("data", (chunk) => {
          rawEmail += chunk.toString("utf8");
        });

        stream.on("end", () => {
          simpleParser(rawEmail)
            .then((parsed) => {
              const data = {
                from: parsed.from.value[0],
                to: parsed.to.value[0],
                subject: parsed.subject,
                date: attributes.date,
                attachmentStatus: parsed.attachments.length > 0,
                date: parsed.date,
                mailId: attributes.uid,
                flags: attributes.flags,
                messageId: parsed.messageId,
                references: parsed.references || [],
                inReplyTo: parsed.inReplyTo || [],
                folder: folder,
                body: parsed.text.slice(0, 100),
              };
              resolve(data);
            })
            .catch((parseErr) => {
              console.error("Error parsing email:", parseErr);
              reject(parseErr);
            });
        });
      });

      msg.on("attributes", (attrs) => {
        attributes = attrs;
      });
    });

    f.once("error", (fetchErr) => reject(`Error fetching emails: ${fetchErr}`));
  });
};
